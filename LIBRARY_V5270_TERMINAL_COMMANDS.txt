1) EXTRACT RELEASE BUNDLE AND PUSH TO GITHUB

cd ~/Downloads

rm -rf "$HOME/Downloads/sc-library-v5.27.0-release"

BUNDLE="$(find "$HOME/Downloads" -maxdepth 1 -type f \
  -name 'sustainable-catalyst-library-v5.27.0-release-bundle*.zip' \
  -print | head -n 1)"

echo "Using: $BUNDLE"

unzip -q "$BUNDLE" \
  -d "$HOME/Downloads/sc-library-v5.27.0-release"

cd "$HOME/Downloads/sc-library-v5.27.0-release"

chmod +x install_and_push_sustainable_catalyst_library_v5_27_0_macos.sh

./install_and_push_sustainable_catalyst_library_v5_27_0_macos.sh \
  "$HOME/Downloads/sustainable-catalyst-library"

2) COPY BACKEND v2.38.0 TO CONTABO

  -o IdentitiesOnly=yes \
  sustainable-catalyst-library-backend-v2.38.0.zip \
  upgrade_library_backend_v2_38_0_contabo.sh \
  catalystadmin@94.72.113.77:/tmp/

3) SSH TO CONTABO

ssh -i ~/.ssh/id_ed25519 \
  -o IdentitiesOnly=yes \
  catalystadmin@94.72.113.77

4) DEPLOY BACKEND ON CONTABO

./upgrade_library_backend_v2_38_0_contabo.sh \
  /tmp/sustainable-catalyst-library-backend-v2.38.0.zip
